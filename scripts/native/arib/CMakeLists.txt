cmake_minimum_required(VERSION 3.28)
project(liuguang_arib LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ARIBCC_USE_EMBEDDED_FREETYPE ON CACHE BOOL "" FORCE)
set(ARIBCC_USE_FONTCONFIG OFF CACHE BOOL "" FORCE)
set(ARIBCC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARIB_SOURCE}" aribcaption)
add_library(liuguang_arib SHARED arib_jni.cpp)
target_compile_features(liuguang_arib PRIVATE cxx_std_17)
target_link_libraries(liuguang_arib PRIVATE aribcaption log)
target_link_options(liuguang_arib PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
